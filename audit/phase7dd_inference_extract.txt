class UtilityRouter(nn.Module):
    def __init__(self,n):
        super().__init__()
        self.net=nn.Sequential(nn.Linear(n,32),nn.ReLU(),nn.Dropout(0.10),
                               nn.Linear(32,16),nn.ReLU(),nn.Linear(16,1))
    def forward(self,x): return self.net(x).squeeze(1)

nb=torch.load(NEURAL_PATH,map_location="cpu",weights_only=False)
if int(nb.get("best_epoch",-1))!=1 or int(nb.get("expert_seed",-1))!=42:
    raise RuntimeError("Frozen neural router provenance mismatch.")
neural=UtilityRouter(len(nb["feature_names"]))
neural.load_state_dict(nb["state_dict"],strict=True); neural.eval()
nmean=np.asarray(nb["mean"],np.float32); nstd=np.asarray(nb["std"],np.float32)
FEATURE_NAMES=list(nb["feature_names"])

def logits(df,branch):
    return df[[f"logit{k}_{branch}" for k in range(4)]].to_numpy(np.float32)

def build_features(df):
    x=pd.DataFrame(index=df.index)
    for k in range(4):
        p=df[f"p{k}_post"].astype(float); q=df[f"p{k}_pair"].astype(float)
        x[f"post_p{k}"]=p; x[f"pair_p{k}"]=q
        x[f"diff_p{k}"]=q-p; x[f"absdiff_p{k}"]=(q-p).abs()
    for c in ["confidence_post","confidence_pair","entropy_post","entropy_pair","severity_post","severity_pair"]:
        x[c]=df[c].astype(float)
    x["conf_diff"]=df.confidence_pair.astype(float)-df.confidence_post.astype(float)
    x["entropy_diff"]=df.entropy_pair.astype(float)-df.entropy_post.astype(float)
    x["severity_diff"]=df.severity_pair.astype(float)-df.severity_post.astype(float)
    x["pred_disagree"]=(df.pred_pair.to_numpy(int)!=df.pred_post.to_numpy(int)).astype(float)
    if list(x.columns)!=FEATURE_NAMES:
        raise RuntimeError("Final router feature names/order do not match frozen neural bundle.")
    return x

def prob_class1(model,X):
    pr=model.predict_proba(X)
    cls=np.asarray(model.classes_)
    pos=np.where(cls==1)[0]
    if len(pos)!=1: raise RuntimeError("Router classifier has no unique class 1 probability.")
    return pr[:,pos[0]]

def route_all(df):
    X=build_features(df)
    lp=logits(df,"post"); lt=logits(df,"pair")
    ulog=prob_class1(logistic,X)
    uhgb=prob_class1(hgb,X)
    zn=(X.to_numpy(np.float32)-nmean)/nstd
    with torch.inference_mode(): uneu=torch.sigmoid(neural(torch.tensor(zn))).numpy()
    pred={
        "POST-only":df.pred_post.to_numpy(int),
        "Siamese":df.pred_pair.to_numpy(int),
        "Equal 0.5 logit blend":(lp+0.5*(lt-lp)).argmax(1),
        "Static alpha blend":(lp+ALPHA*(lt-lp)).argmax(1),
        "Max-confidence selector":np.where(df.confidence_pair.to_numpy(float)>df.confidence_post.to_numpy(float),
                                           df.pred_pair.to_numpy(int),df.pred_post.to_numpy(int)),
        "Min-entropy selector":np.where(df.entropy_pair.to_numpy(float)<df.entropy_post.to_numpy(float),
                                        df.pred_pair.to_numpy(int),df.pred_post.to_numpy(int)),
        "Logistic soft gate":(lp+ulog[:,None]*(lt-lp)).argmax(1),
        "HGB hard router":np.where(uhgb>=0.5,df.pred_pair.to_numpy(int),df.pred_post.to_numpy(int)),
        "Neural Temporal Utility Router":(lp+uneu[:,None]*(lt-lp)).argmax(1),
    }
    y=df.true_label.to_numpy(int)
    pred["Oracle expert selector"]=np.where(
        (df.pred_pair.to_numpy(int)==y)&(df.pred_post.to_numpy(int)!=y),
        df.pred_pair.to_numpy(int),df.pred_post.to_numpy(int))
    return pred,{"logistic":ulog,"hgb":uhgb,"neural":uneu}

Pclean,Gclean=route_all(clean)
