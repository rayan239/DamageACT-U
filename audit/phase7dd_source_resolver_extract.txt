# ---------- filesystem + artifact helpers ----------
def sha256_bytes(b):
    return hashlib.sha256(b).hexdigest()

def sha256_file(p):
    h=hashlib.sha256()
    with open(p,"rb") as f:
        for chunk in iter(lambda:f.read(1024*1024),b""): h.update(chunk)
    return h.hexdigest()

if nested.exists():

dataset_roots.extend([x.resolve() for x in INPUT_ROOT.iterdir() if x.is_dir() and x.name!="datasets"])

# Exclude the huge xBD dataset wrapper from recursive artifact searches.
artifact_roots=[]
for ds in dataset_roots:
    try:
        if XBD_ROOT==ds or XBD_ROOT.is_relative_to(ds):
            continue
    except Exception:
        if str(XBD_ROOT).startswith(str(ds)):
            continue
    artifact_roots.append(ds)

_zip_name_cache={}
def zip_names(zp):
    key=str(zp)
    if key not in _zip_name_cache:
        try:
            with zipfile.ZipFile(zp,"r") as z: _zip_name_cache[key]=z.namelist()
        except zipfile.BadZipFile: _zip_name_cache[key]=[]
    return _zip_name_cache[key]

def candidates_for_suffix(suffix):
    out=[]
    base=Path(suffix).name
    for root in artifact_roots:
        for p in root.rglob(base):
            if p.is_file() and p.as_posix().endswith(suffix):
                out.append({"mode":"direct","path":p,"member":None,"source":str(p)})
        for zp in root.rglob("*.zip"):
            for m in zip_names(zp):
                if m.replace("\\","/").endswith(suffix):
                    out.append({"mode":"zip","path":zp,"member":m,"source":f"{zp}::{m}"})
    return sorted(out,key=lambda x:x["source"])

def raw_bytes(c):
    if c["mode"]=="direct": return c["path"].read_bytes()
    with zipfile.ZipFile(c["path"],"r") as z: return z.read(c["member"])

def materialize(c,name):
    dst=CACHE/name
    dst.write_bytes(raw_bytes(c))
    return dst

def json_candidates(suffix):
    vals=[]
    for c in candidates_for_suffix(suffix):
        try: vals.append((c,json.loads(raw_bytes(c).decode("utf-8-sig"))))
        except Exception: pass
    return vals

def choose_json(suffix,predicate,label):
    hits=[(c,j) for c,j in json_candidates(suffix) if predicate(j)]
    if not hits: raise RuntimeError(f"Missing valid {label} ({suffix}).")
    # Duplicate byte-identical copies are harmless; conflicting valid copies are not.
    hashes={sha256_bytes(raw_bytes(c)) for c,_ in hits}
    if len(hashes)>1:
        raise RuntimeError(f"Multiple conflicting valid {label} files are attached. Remove duplicate historical packages.")
    return hits[0]

def choose_bytes_with_sha(suffix,expected_sha,label):
    hits=[]
    for c in candidates_for_suffix(suffix):
        b=raw_bytes(c)
        if sha256_bytes(b)==expected_sha: hits.append(c)
    if not hits: raise RuntimeError(f"Could not locate {label} with expected SHA256 {expected_sha}.")
    return hits[0]

print("PASS: all four frozen project source hashes match.")

def subset_dir(subset):
    aliases={
        "train":["train","tier1"],"tier1":["tier1","train"],
        "tier3":["tier3"],"test":["test"],
        "hold":["hold","holdout"],"holdout":["holdout","hold"],
    }.get(str(subset),[str(subset)])

    if not good: raise RuntimeError(f"No physical xBD subset directory for manifest subset={subset}")
    return good[0]
